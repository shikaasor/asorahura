### `src/components/automate/OfferingCard.tsx` (presentational component, static rendering) — NEW

**Component structure** (PricingSection.tsx lines 1-25, adapted):
```typescript
"use client";

import Link from "next/link";
import type { OfferingMetadata } from "@/lib/checkout";
import { trackAnalyticsEvent } from "@/lib/analytics";
import styles from "./OfferingCard.module.css";

interface OfferingCardProps {
  offering: OfferingMetadata;
}

export default function OfferingCard({ offering }: OfferingCardProps) {
  const handleClick = () => {
    trackAnalyticsEvent("Demo Interaction", { offering_id: offering.id });
  };

  return (
    <Link href={`/automate/${offering.id}`} onClick={handleClick} className={styles.card}>
      <div className={styles.header}>
        <h3 className={styles.title}>{offering.name}</h3>
        {offering.tagline && <p className={styles.tagline}>{offering.tagline}</p>}
      </div>

      <div className={styles.body}>
        <p className={styles.description}>{offering.description}</p>

        <div className={styles.pricing}>
          {offering.setupPrice && (
            <>
              <p className={styles.price}>${(offering.setupPrice / 100).toFixed(0)}</p>
              <p className={styles.label}>{offering.setupLabel}</p>
            </>
          )}
          {offering.monthlyPrice && (
            <>
              <p className={styles.price}>${(offering.monthlyPrice / 100).toFixed(2)}/mo</p>
              <p className={styles.label}>monthly</p>
            </>
          )}
        </div>
      </div>

      <div className={styles.footer}>
        <button className={styles.cta}>{offering.checkoutCTA}</button>
      </div>
    </Link>
  );
}
```
